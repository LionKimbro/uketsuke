from . import paths, state


def initialize_program_at_startup_time_once(new_configuration):
    state.initialize_configuration_when_the_program_starts_up(new_configuration)
    paths.ensure_that_the_necessary_directories_exist_when_the_program_starts_up()
    repair_system_from_potentially_crashed_state()


def repair_system_from_potentially_crashed_state():
    heal_incomplete_job_transition()
    heal_interrupted_execution_job()


def heal_incomplete_job_transition():
    (purpose: resolve the residue left by one specific interrupted transition)

    authoritative_job_file = paths.path("job")
    staged_replacement_job_file = paths.path("job-nextstate")
    
    look for staged_replacement_job_file
    
    if no staged replacement exists:
        return
    
    if it is not completely parsable JSON:
        delete it
	(durably sync the containing directory)
        return

    copy staged_replacement_job_file to/over authoritative_job_file
    (sync the authoritative job file)
    
    delete the now-unnecessary staged replacement job file
	(durably sync the containing directory)


def heal_interrupted_execution_job():
    read the current active job, if there is one

    if there is no active job:
        return

    if the job status is not "execution-attempted":
        return

    reset the job status to "claimed"
    append a durable history entry reporting the interrupted execution
    and that the job is prepared for re-execution
    (the durable history append also durably saves the reset job state)

NOTE 1:
We do not rely on os.replace(staged, authoritative) because it is not durable!
"...and because ordinary Python os.replace() does not request MOVEFILE_WRITE_THROUGH, its return does not by itself establish that the rename metadata survived a sudden power loss. Microsoft exposes MOVEFILE_WRITE_THROUGH specifically to delay return until the operation has been flushed."
So, instead, we (A) copy-with-sync, (B) delete-with-sync.

NOTE 2:
The host API will supply functions for copy-and-sync, and delete-and-sync, so this will be one instruction.
